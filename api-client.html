/** * API CLIENT * Cliente para Google Apps Script usando google.script.run * Compatível com HTML Service do GAS */ class APIClient { constructor() { this.token = null; this.interceptors = { request: [], response: [], error: [] }; this.cache = new Map(); this.cacheTTL = 300000; // 5 min default } /** * Set authentication token */ setToken(token) { this.token = token; if (token) { localStorage.setItem('canabica_token', token); } else { localStorage.removeItem('canabica_token'); } } /** * Get stored token */ getToken() { return this.token || localStorage.getItem('canabica_token'); } /** * Add request interceptor */ addRequestInterceptor(callback) { this.interceptors.request.push(callback); } /** * Add response interceptor */ addResponseInterceptor(callback) { this.interceptors.response.push(callback); } /** * Add error interceptor */ addErrorInterceptor(callback) { this.interceptors.error.push(callback); } /** * Generate UUID v4 for correlationId */ _generateUUID() { return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, function(c) { var r = Math.random() * 16 | 0, v = c == 'x' ? r : (r & 0x3 | 0x8); return v.toString(16); }); } /** * Call backend via google.script.run */ async callBackend(action, data) { const token = this.getToken(); const context = { action: action, data: data || {}, token: token, correlationId: this._generateUUID() }; // Request interceptors for (const interceptor of this.interceptors.request) { await interceptor(context); } return new Promise((resolve, reject) => { google.script.run .withSuccessHandler((response) => { // Response interceptors Promise.all(this.interceptors.response.map(i => i(response))) .then(() => { if (response && response.success) { resolve(response); } else { const error = new APIError( (response && response.error && response.error.message) || 'Erro desconhecido', response && response.statusCode, response ); // Error interceptors Promise.all(this.interceptors.error.map(i => i(error))) .finally(() => reject(error)); } }); }) .withFailureHandler((error) => { const apiError = new APIError(error.message || 'Falha na chamada ao backend', 500, error); // Error interceptors Promise.all(this.interceptors.error.map(i => i(apiError))) .finally(() => reject(apiError)); }) .routeRequest(context); }); } /** * Refresh token automatically */ async refreshToken() { try { const response = await this.callBackend('auth.refresh', {}); if (response.data && response.data.token) { this.setToken(response.data.token); return response.data.token; } } catch (error) { console.error('Token refresh failed:', error); throw error; } } /** * Generic request wrapper */ async request(action, data = null) { return this.callBackend(action, data); } /** * GET-like request (list/read operations) */ async get(action, data = null) { // Check cache const cacheKey = action + JSON.stringify(data); const cached = this.cache.get(cacheKey); if (cached && Date.now() - cached.timestamp < this.cacheTTL) { return cached.data; } const response = await this.callBackend(action, data); // Store in cache this.cache.set(cacheKey, { data: response, timestamp: Date.now() }); return response; } /** * POST-like request (create operations) */ async post(action, data) { return this.callBackend(action, data); } /** * PUT-like request (update operations) */ async put(action, data) { return this.callBackend(action, data); } /** * DELETE-like request (delete operations) */ async delete(action, data) { return this.callBackend(action, data); } /** * Clear cache */ clearCache() { this.cache.clear(); } /** * Check if authenticated */ isAuthenticated() { return !!this.getToken(); } } /** * API Error */ class APIError extends Error { constructor(message, status, data) { super(message); this.status = status; this.data = data; this.name = 'APIError'; } } /** * Global API client instance */ const api = new APIClient(); /** * Setup interceptors */ // Auto-refresh token if 401 // Backend v2.0 retorna tanto error.status quanto error.data.statusCode api.addErrorInterceptor(async (error) => { if (error.status === 401 || (error.data && error.data.statusCode === 401)) { console.warn('Token expirado ou inválido'); api.setToken(null); window.dispatchEvent(new CustomEvent('auth-expired')); } }); // Log errors api.addErrorInterceptor(async (error) => { console.error('API Error:', error); }); // ============================================================================ // API METHODS — DOMÍNIO DE QUALIDADE DE INSUMOS (PROMPT 2) // ============================================================================ /** * INSUMOS */ /** * Cria novo insumo * @param {Object} data - Dados do insumo * @param {string} data.nomeComercial - Nome comercial (obrigatório) * @param {string} data.tipoSal - Tipo de sal (obrigatório) * @param {string} data.fabricante - Fabricante (obrigatório) * @param {string} [data.paisOrigem] - País de origem * @param {string} [data.registroMAPAEstabelecimento] - Registro MAPA estabelecimento * @param {string} [data.registroMAPAProduto] - Registro MAPA produto * @param {string} [data.agenteQuelante] - EDTA, DTPA ou EDDHA * @param {number} [data.solubilidade_gL_20C] - Solubilidade em g/L a 20°C * @param {number} [data.purezaPercentual] - Pureza em % (0-100) * @param {string} [data.driveFichaTecnicaURL] - URL da ficha técnica * @param {string} [data.ativo] - ATIVO ou INATIVO * @returns {Promise
